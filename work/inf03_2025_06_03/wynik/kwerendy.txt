-- Zapytanie 1
SELECT nazwisko, imie, klasa, data_wypozyczenia FROM uczniowie WHERE data_wypozyczenia > '2020-03-02';
-- Zapytanie 2
SELECT uczniowie.nazwisko, uczniowie.imie, uczniowie.data_wypozyczenia, ksiazki.autor, ksiazki.kod FROM uczniowie JOIN ksiazki ON uczniowie.id_ksiazki = ksiazki.id;
-- Zapytanie 3
DELETE FROM ksiazki WHERE autor = '0';
-- Zapytanie 4
SELECT autor, tytul, kod FROM ksiazki ORDER BY RAND() LIMIT 5;
