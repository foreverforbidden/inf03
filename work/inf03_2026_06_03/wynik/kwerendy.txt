-- Zapytanie 1
SELECT nazwa, styl, cena FROM meble WHERE czyNowy = 1 AND cena <= 1000 ORDER BY cena ASC;
-- Zapytanie 2
CREATE VIEW komody AS SELECT idMeble, nazwa, kategoria, cena FROM meble WHERE kategoria = 3;
-- Zapytanie 3
SELECT ROUND(AVG(cena)) AS `Średnia cena komód`, MIN(cena) AS `Minimalna`, MAX(cena) AS `Maksymalna` FROM komody;
-- Zapytanie 4
DELETE FROM klienci WHERE imie = 'Tomasz' AND nazwisko = 'Nowak';
