-- Zapytanie 1
SELECT tytul, gatunek FROM gry WHERE YEAR(data_wydania) BETWEEN 2018 AND 2023 AND gatunek LIKE '%Niezależne%';
-- Zapytanie 2
SELECT producenci.nazwa, COUNT(gry.id_gry) AS liczba_gier FROM producenci LEFT JOIN gry ON producenci.id_producenta = gry.id_producenta GROUP BY producenci.nazwa;
-- Zapytanie 3
SELECT gry.id_gry, gry.tytul, producenci.nazwa AS producent, wydawcy.nazwa AS wydawca FROM gry JOIN producenci ON gry.id_producenta = producenci.id_producenta JOIN wydawcy ON gry.id_wydawcy = wydawcy.id_wydawcy;
-- Zapytanie 4
ALTER TABLE gry ADD ocena INT NOT NULL DEFAULT 5;
