SELECT nazwa FROM choroby WHERE zakazna = 'T' ORDER BY nazwa ASC;
-- Zapytanie 2
SELECT id, nazwa FROM choroby;
-- Zapytanie 3
SELECT objawy.nazwa FROM objawy JOIN choroby_objawy ON objawy.id = choroby_objawy.id_objawy WHERE choroby_objawy.id_choroby = 5;
-- Zapytanie 4
SELECT choroby.nazwa, COUNT(choroby_objawy.id_objawy) FROM choroby JOIN choroby_objawy ON choroby.id = choroby_objawy.id_choroby WHERE choroby.zakazna = 'T' GROUP BY choroby.id, choroby.nazwa;
